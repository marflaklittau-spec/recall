🚀 部署与配置指引 ›
  1. 创建仓库：在 GitHub 新建一个 公开 仓库，把 index.html 和 supabase.js 上传进去。
  2. 修改配置：编辑 index.html 中的 CONFIG.repo（源码仓库，用于 Star 按钮和问题反馈）；在 Supabase 免费建项目后，把 URL 和 anon key 填入 CONFIG.supabaseUrl / supabaseKey（账号登录与回忆墙都依赖它），并把 CONFIG.adminEmail 设为站长邮箱（该账号可删除任何照片和留言）。
  3. 开启 Pages：仓库 Settings → Pages → Source 选择 main 分支 → Save。约 1 分钟后通过 https://用户名.github.io/仓库名/ 访问。
  4. 初始化云数据：在 Supabase 项目的 SQL Editor 中运行以下语句，创建回忆墙数据表与公开存储桶：
create table wall (
  id uuid primary key default gen_random_uuid(),
  title text default '', author text default '',
  note text default '', mono_public boolean default true,
  date timestamptz default now(), storage_path text not null,
  owner uuid references auth.users(id), created_at timestamptz default now()
);
alter table wall enable row level security;
create policy "wall_read" on wall for select using (true);
create policy "wall_write" on wall for insert to authenticated with check (true);
create policy "wall_edit" on wall for update to authenticated using (auth.uid() = owner);
create table wall_monos (
  wall_id uuid primary key references wall(id) on delete cascade,
  mono text default ''
);
alter table wall_monos enable row level security;
create policy "wm_read" on wall_monos for select using (
  exists (select 1 from wall w where w.id = wall_monos.wall_id and (w.mono_public or w.owner = auth.uid()))
);
create policy "wm_write" on wall_monos for insert to authenticated with check (
  exists (select 1 from wall w where w.id = wall_id and w.owner = auth.uid())
);
create policy "wm_edit" on wall_monos for update to authenticated using (
  exists (select 1 from wall w where w.id = wall_monos.wall_id and w.owner = auth.uid())
);
create table comments (
  id uuid primary key default gen_random_uuid(),
  wall_id uuid references wall(id) on delete cascade,
  author text default '匿名', text text not null,
  created_at timestamptz default now()
);
alter table comments enable row level security;
create policy "cmt_read" on comments for select using (
  exists (select 1 from wall w where w.id = comments.wall_id and (w.mono_public or w.owner = auth.uid()))
);
create policy "cmt_write" on comments for insert to authenticated with check (
  exists (select 1 from wall w where w.id = wall_id and w.mono_public)
);
insert into storage.buckets (id, name, public) values ('photos','photos',true);
create policy "img_read" on storage.objects for select using (bucket_id = 'photos');
create policy "img_write" on storage.objects for insert to authenticated with check (bucket_id = 'photos');
-- 删除权限：管理员邮箱可删任何内容，上传者可删自己的照片
create policy "wall_delete" on wall for delete to authenticated
  using (auth.jwt()->>'email' = '你的管理员邮箱@example.com' or auth.uid() = owner);
create policy "cmt_delete" on comments for delete to authenticated
  using (auth.jwt()->>'email' = '你的管理员邮箱@example.com');
create policy "img_delete" on storage.objects for delete to authenticated
  using (bucket_id = 'photos' and (auth.jwt()->>'email' = '你的管理员邮箱@example.com'
    or exists (select 1 from wall w where w.storage_path = storage.objects.name and w.owner = auth.uid())));

💡 回忆墙照片存储在 Supabase 云端（免费层 1GB，约数千张压缩照片）。登录用户在站内即可直接上传上墙、编写独白、发表留言，无需 GitHub 账号；未登录访客可自由浏览。网站源码存放在 GitHub 仓库，可随时查看和备份。站长（CONFIG.adminEmail 对应账号）可删除任何不当照片与留言，上传者也可删除自己的照片。
